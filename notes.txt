closure is a function along with its outer enviromne tis known as closure (in other words a function along with its lexical scope is known as closure )


function outer(){
    let a=10
    function inner(){
        console.log(a)
    }
    return inner
}

outer()();




function outer(b){
    let a=10
    function inner(){
        console.log(a,b)
    }
    return inner
}

outer("Hello world")();

this inner will also have accesss to the "b" variable



closure
